n, s, f = map(int, input().split())
mas = []
s -= 1
f -= 1
for i in range(n):
    mas.append([int(i) for i in input().split()])
u = [0] * n
v = [999999] * n
v[s] = 0

for i in range(n):
    min = 999999
    ind = 0
    for j in range(n):
        if v[j] < min and u[j] == 0:
            min = v[j]
            ind = j
    for j in range(n):
        if mas[ind][j] > 0:
            if v[ind] + mas[ind][j] < v[j]:
                v[j] = v[ind] + mas[ind][j]
    u[ind] = 1
if v[f] == 999999:
    print(-1)
else:
    print(v[f])
